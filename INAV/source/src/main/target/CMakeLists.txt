file(GLOB entries *)
foreach(subdir ${entries})
    set(entry "${subdir}/CMakeLists.txt")
    get_filename_component(target_name ${subdir} NAME)
    if (DEFINED INAV_TARGET_DIRS AND NOT target_name IN_LIST INAV_TARGET_DIRS)
        continue()
    endif()
    if (EXISTS ${entry})
        add_subdirectory(${subdir})
    endif()
endforeach()
